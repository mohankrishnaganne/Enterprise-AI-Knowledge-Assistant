# =============================================================================
# Runtime dependencies — what the agent, API and UI need to serve requests.
#
# Development and evaluation extras (RAGAS, pytest, ruff, reportlab) live in
# requirements-dev.txt, which installs this file first. Keeping them apart matters
# for deployment: Streamlit Community Cloud installs THIS file, and the evaluation
# stack would add hundreds of megabytes it never uses.
#
# Targets the LangChain/LangGraph 1.x line (current stable), NOT the 0.2/0.3 line
# most RAG tutorials still use.
#
# ALWAYS install into a virtualenv:
#     python -m venv .venv && .venv/Scripts/activate      # Windows
#     python -m venv .venv && source .venv/bin/activate   # macOS / Linux
#
# The extra index below makes pip resolve the CPU-only torch build. The default
# PyPI wheel bundles CUDA and pulls ~2.5GB of nvidia packages that are useless on
# a CPU host and too large for most free tiers.
# =============================================================================

--extra-index-url https://download.pytorch.org/whl/cpu
